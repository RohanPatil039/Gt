import numpy as np
import nashpy as nash



# Player A's payoff matrix
A = np.array([[1, 3],
              [0, 2]])

# Player B's payoff matrix
B = np.array([[1, 0],
              [3, 2]])

# Create the game
game = nash.Game(A, B)

print("Game Matrix (Payoffs for Player A and Player B):")
print(game)

# Compute mixed strategy Nash equilibria
equilibria = list(game.support_enumeration())

print("\nMixed Strategy Nash Equilibria:")
for eq in equilibria:
    print(f"Player A Strategy: {eq[0]}")
    print(f"Player B Strategy: {eq[1]}")
