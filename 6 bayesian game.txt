import numpy as np

# Type probabilities for Player 1's types
type_probs = {"T1": 0.6, "T2": 0.4}

# Payoff matrices for each type of Player 1
payoff_T1 = np.array([[3, 0], 
                      [5, 1]])
payoff_T2 = np.array([[2, 4], 
                      [1, 3]])

# Compute expected payoff matrix for Player 1
expected_payoff = (type_probs["T1"] * payoff_T1) + (type_probs["T2"] * payoff_T2)

# Display expected payoff matrix
print("Expected Payoff Matrix for Player 1 (Bayesian Game):")
print(np.round(expected_payoff, 2))

# Minimax-like decision rule: pick strategy with best worst-case payoff
row_minima = np.min(expected_payoff, axis=1)
best_strategy_index = np.argmax(row_minima)

# Print best strategy based on expected payoff
print(f"\nBest strategy for Player 1 (Bayesian): Strategy {['A', 'B'][best_strategy_index]}")
